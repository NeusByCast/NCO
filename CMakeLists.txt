cmake_minimum_required(VERSION 3.25)
project(NCO VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(nco-core-tests tests/core_tests.cpp)
target_include_directories(nco-core-tests PRIVATE src)
add_test(NAME core COMMAND nco-core-tests)
add_executable(nco-audio-tests tests/audio_tests.cpp)
target_include_directories(nco-audio-tests PRIVATE src)
add_test(NAME audio COMMAND nco-audio-tests)
if(NOT WIN32)
  return()
endif()
enable_language(RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
find_package(FFMPEG REQUIRED)
add_executable(NCO WIN32 src/main.cpp src/ui.cpp src/ui_test.cpp src/config.cpp src/capture.cpp src/pointer.cpp src/audio.cpp src/media.cpp src/self_test.cpp src/platform_test.cpp src/engine.cpp src/platform.cpp src/diagnostics.cpp assets/nco.rc)
target_include_directories(NCO PRIVATE src ${FFMPEG_INCLUDE_DIRS})
target_link_directories(NCO PRIVATE ${FFMPEG_LIBRARY_DIRS})
target_compile_definitions(NCO PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(NCO PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_libraries(NCO PRIVATE ${FFMPEG_LIBRARIES} d3d11 dxgi d2d1 dwrite dwmapi windowsapp runtimeobject ole32 oleaut32 uuid avrt propsys psapi shell32 shlwapi comdlg32 comctl32 mfplat mfuuid strmiids bcrypt secur32 ws2_32)
target_link_options(NCO PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/assets/nco.manifest)
install(TARGETS NCO RUNTIME DESTINATION .)
